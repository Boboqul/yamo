# ЯМО 1.0 — руководство языка

Точка входа: `функция главная() { ... }`. Запуск: `yamo файл.ямо` или `yamo` (REPL).

## Типы и переменные
    целое n = 5;  дробное x = 1.5;  текст s = "привет";  логическое b = истина;
    целое[] nums = [1, 2, 3];         :: массив
    словарь d = json_разобрать("{}"); :: словарь (ключ → значение)

## Операторы
    + - * / %   == != < > <= >=   и или не

## Условия и циклы
    если (x > 1) { ... } иначе { ... }
    для (целое i = 0; i < 10; i = i + 1) { ... }
    пока (условие) { ... }
    прервать; продолжить;

## Функции
    функция сумма(целое a, целое b) { вернуть a + b; }

## Классы и наследование
    класс Человек {
        текст имя;
        Человек(текст n) { этот.имя = n; }
        функция привет() { вывод "Привет, " + этот.имя; }
    }
    класс Студент : Человек {
        текст вуз;
        Студент(текст n, текст u) { этот.имя = n; этот.вуз = u; }
    }
    Человек p = новый Человек("Али");
    p.привет();

## Исключения
    попробовать { ошибка("упс"); } поймать (е) { вывод "Поймано: " + е; }

## Модули и пакеты
    импорт "stdlib.ямо";                      :: локальный модуль
    импорт_из_сети("http://host/mod.ямо");    :: скачать и подключить (кеш)

## Файлы
    читать_файл, запись_файл, дописать_файл, существует_файл, удалить_файл

## JSON и словари
    словарь d = json_разобрать(текст);  текст t = json_собрать(d);
    ключи(d); значения(d); содержит_ключ(d, "ключ"); d["ключ"] = 5;

## Сеть
    текст html = http_получить(url);
    текст otvet = http_отправить(url, тело, "POST", токен);

## GUI (Windows)
    сообщение("текст"); сообщение_с_заголовком("заголовок", "текст");
    логическое da = вопрос("Да/Нет?"); текст s = ввод_окно("Заголовок", "Подсказка:");

## База данных (SQLite)
    бд_открыть("my.db"); бд_запрос("CREATE TABLE t(a INTEGER)");
    бд_запрос("INSERT INTO t VALUES (1)");
    строки = бд_выбрать("SELECT a FROM t");   :: массив строк (массивов)
    бд_закрыть();

## REPL
    ymo> 2+2*3        :: выражения печатаются сами
    8
    ymo> текст имя = "ЯМО"   :: объявления запоминаются между строками
    ymo> выход
